/**
 * @file PothosCareDevice.ino
 * @brief Sistema IoT para el cuidado de una planta Pothos.
 *        Framework: ModestIoT + ArduinoJson. Autenticación JWT sin Edge API Key.
 *        Compatible con Wokwi y prototipo físico real.
 *
 * Arquitectura de red (corrige el error "conexión: -1"):
 *   - TODA la E/S de red (login, sync, telemetría, reporte) ocurre EXCLUSIVAMENTE
 *     en loop() (tarea principal, stack grande), de forma secuencial. Nunca se
 *     hacen peticiones HTTPS desde la tarea de muestreo de sensores.
 *   - El manejador de eventos on() solo actualiza actuadores por GPIO y marca
 *     banderas; loop() consume esas banderas para enviar los reportes.
 *   - Se reutiliza un único WiFiClientSecure para reducir handshakes TLS.
 *
 * Umbrales: se sincronizan periódicamente desde el backend (/sync/config) y los
 * actuadores reaccionan cuando las lecturas cruzan esos límites.
 */

/**
 * @brief Sensor LDR que devuelve porcentaje de luminosidad (0-100).
 */
class LdrSensor : public Sensor {
private:
    int cachedLightPercent;
    int lastRawValue;
protected:
    void processEvent(Event& event) override {
        if (event.identifier == MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER) {
            int currentRaw = analogRead(pin);
            if (abs(currentRaw - lastRawValue) > 50) {
                lastRawValue = currentRaw;
                cachedLightPercent = map(currentRaw, 0, 4095, 0, 100);
                event = Event(DATA_READ_EVENT_IDENTIFIER, pin);
            } else {
                event.identifier = -1;
            }
        }
    }
public:
    LdrSensor(int signalGpioPin, EventHandler* parentHandler = nullptr)
        : Sensor(signalGpioPin, parentHandler), cachedLightPercent(0), lastRawValue(0) {
        pinMode(pin, INPUT);
    }
    int getLightPercent() const { return cachedLightPercent; }
};

/**
 * @brief Sensor de gas analógico que devuelve porcentaje (0-100).
 */
class AnalogGasSensor : public Sensor {
private:
    int cachedGasPercent;
    int lastRawValue;
protected:
    void processEvent(Event& event) override {
        if (event.identifier == MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER) {
            int currentRaw = analogRead(pin);
            if (abs(currentRaw - lastRawValue) > 50) {
                lastRawValue = currentRaw;

                event = Event(DATA_READ_EVENT_IDENTIFIER, pin);
            } else {
                event.identifier = -1;
            }
        }
    }
public:

// ---------------------------------------------------------------------------
// Clase mediadora del dispositivo
// ---------------------------------------------------------------------------
class PothosCareDevice : public Device {
public:
    enum class BuzzerMode { ON, OFF };
    enum class ActuatorMode { ON, OFF, AUTO };

private:
    DhtSensor dhtSensor;
    AnalogGasSensor gasSensor;
    LdrSensor ldrSensor;

    Button btnBuzzer;
    Button btnServo;
    Button btnLed;

    Led uvRelay;
    ServoMotor waterValve;
    Buzzer alarmBuzzer;

    CharacterLcdDisplay statusDisplay1;
    CharacterLcdDisplay statusDisplay2;

    // Modos deseados (desde backend o botones)
    BuzzerMode currentBuzzerMode;
    ActuatorMode currentServoMode;
    ActuatorMode currentLedMode;

    // Estado físico real (leído desde loop() para reportar; escrito desde on())
    volatile bool physicalBuzzerOn;
    volatile bool physicalServoOn;
    volatile bool physicalLedOn;

    // Bandera de reporte pendiente: on() la activa, loop() la consume y envía el PUT.
    volatile bool actuatorReportPending;

    // Umbrales cacheados (un único umbral de luz: mínimo)
    float cachedTempMin;
    float cachedTempMax;
    float cachedHumidityMin;
    float cachedHumidityMax;
    float cachedLightMin;

    /**
     * @brief Evalúa reglas de negocio y actualiza actuadores físicos (solo GPIO).
     *        NO hace red: si hubo cambio físico, marca actuatorReportPending para
     *        que loop() envíe el reporte. Seguro de llamar desde cualquier tarea.
     */
    void evaluateBusinessRules() {
        float tempC = dhtSensor.getTemperatureInCelsius();
        float humidity = dhtSensor.getRelativeHumidityInPercentage();
        int gasPct = gasSensor.getGasPercent();
        int lightPct = ldrSensor.getLightPercent();

        bool changed = false;

        // ---- Servo / Válvula de agua: riega si la humedad cae por debajo del mínimo ----
        bool newServo;
        if (currentServoMode == ActuatorMode::ON) {
            waterValve.handle(ServoMotor::LOCK_TO_MAXIMUM_LIMIT_COMMAND);
            newServo = true;
        } else if (currentServoMode == ActuatorMode::OFF) {
            waterValve.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
            newServo = false;
        } else { // AUTO
            if (humidity != INVALID_READING && humidity < cachedHumidityMin) {
                waterValve.handle(ServoMotor::LOCK_TO_MAXIMUM_LIMIT_COMMAND);
                newServo = true;
            } else {
                waterValve.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
                newServo = false;
            }
        }
        if (newServo != physicalServoOn) { physicalServoOn = newServo; changed = true; }

        // ---- LED UV: se enciende cuando FALTA luz (por debajo del umbral mínimo) ----
        bool newLed;
        if (currentLedMode == ActuatorMode::ON) {
            uvRelay.handle(Led::TURN_ON_COMMAND);
            newLed = true;
        } else if (currentLedMode == ActuatorMode::OFF) {
            uvRelay.handle(Led::TURN_OFF_COMMAND);
            newLed = false;
        } else { // AUTO
            if (lightPct < cachedLightMin) {
                uvRelay.handle(Led::TURN_ON_COMMAND);
                newLed = true;
            } else {
                uvRelay.handle(Led::TURN_OFF_COMMAND);
                newLed = false;
            }
        }
        if (newLed != physicalLedOn) { physicalLedOn = newLed; changed = true; }

        // ---- Buzzer: alarma si la temperatura sale del rango o el gas es alto ----
        bool newBuzzer;
        if (currentBuzzerMode == BuzzerMode::OFF) {
            alarmBuzzer.handle(Buzzer::TURN_OFF_COMMAND);
            newBuzzer = false;
        } else { // ON (modo alerta)
            bool tempOutOfRange = (tempC != INVALID_READING) &&
                                  (tempC < cachedTempMin || tempC > cachedTempMax);
            bool gasHigh = gasPct > GAS_ALERT_PERCENT;
            if (tempOutOfRange || gasHigh) {
                alarmBuzzer.handle(Buzzer::TURN_ON_COMMAND);
                newBuzzer = true;
            } else {
                alarmBuzzer.handle(Buzzer::TURN_OFF_COMMAND);
                newBuzzer = false;
            }
        }
        if (newBuzzer != physicalBuzzerOn) { physicalBuzzerOn = newBuzzer; changed = true; }

        if (changed) {
            actuatorReportPending = true;   // loop() enviará el reporte
        }
    }

    /**
     * @brief Envía PUT /actuators/reported con el estado físico actual.
     *        SOLO debe llamarse desde loop() (tarea principal).
     */
    void reportActuatorState() {
        if (authToken == "") return;

        HTTPClient http;
        http.setTimeout(HTTP_TIMEOUT_MS);
        http.begin(secureClient, String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/actuators/reported");
        http.addHeader("Content-Type", "application/json");
        http.addHeader("Authorization", "Bearer " + authToken);

        String buzzerStr = physicalBuzzerOn ? "ON" : "OFF";
        String servoStr = physicalServoOn ? "ON" : "OFF";
        String ledStr = physicalLedOn ? "ON" : "OFF";

        String payload = "{\"buzzerMode\":\"" + buzzerStr
                       + "\",\"servoMode\":\"" + servoStr
                       + "\",\"ledMode\":\"" + ledStr + "\"}";

        int code = http.PUT(payload);
        if (code == 200) {
            Serial.printf("[Reported] PUT /actuators/reported -> %d\n", code);
        } else if (code == 401) {
            Serial.println(F("[Auth] Token expirado en report"));
            reLogin();
        } else if (code > 0) {
            Serial.printf("[Reported] Error servidor: %d\n", code);
        } else {
            Serial.printf("[Reported] Error conexión: %d\n", code);
        }
        http.end();
    }

public:
    PothosCareDevice()
        : Device(SAMPLING_INTERVAL_MS, TIMER_GROUP_INDEX),
          dhtSensor(DHT_PIN, DHT22, this),
          gasSensor(GAS_A0, this),
          ldrSensor(LDR_PIN, this),
          btnBuzzer(BTN1_PIN, 50, this),
          btnServo(BTN2_PIN, 50, this),
          btnLed(BTN3_PIN, 50, this),
          uvRelay(RELAY_PIN, false, this),
          waterValve(SERVO_PIN, 0, this),
          alarmBuzzer(BUZZER_PIN, this),
          statusDisplay1(LCD1_I2C_ADDRESS, 16, 2, true, this),
          statusDisplay2(LCD2_I2C_ADDRESS, 16, 2, true, this),
          currentBuzzerMode(BuzzerMode::ON),
          currentServoMode(ActuatorMode::AUTO),
          currentLedMode(ActuatorMode::AUTO),
          physicalBuzzerOn(false),
          physicalServoOn(false),
          physicalLedOn(false),
          actuatorReportPending(false),
          cachedTempMin(DEFAULT_TEMP_MIN),
          cachedTempMax(DEFAULT_TEMP_MAX),
          cachedHumidityMin(DEFAULT_HUMIDITY_MIN),
          cachedHumidityMax(DEFAULT_HUMIDITY_MAX),
          cachedLightMin(DEFAULT_LIGHT_MIN)
    {
        initializeAsynchronousEngine(ASYNC_ENGINE_QUEUE_DEPTH);

        appendSensorToScheduler(&dhtSensor, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&gasSensor, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&ldrSensor, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&btnBuzzer, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&btnServo, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&btnLed, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);

        statusDisplay1.handle(CharacterLcdDisplay::TURN_BACKLIGHT_ON_COMMAND);
        statusDisplay2.handle(CharacterLcdDisplay::TURN_BACKLIGHT_ON_COMMAND);

        waterValve.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
        uvRelay.handle(Led::TURN_OFF_COMMAND);
        alarmBuzzer.handle(Buzzer::TURN_OFF_COMMAND);

        statusDisplay1.setLineBuffer(0, " Pothos Care IoT");
        statusDisplay1.setLineBuffer(1, " Iniciando...");
        statusDisplay1.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);
    }

    // -----------------------------------------------------------------------
    // Getters públicos para telemetría
    // -----------------------------------------------------------------------
    float getTemperature() const { return dhtSensor.getTemperatureInCelsius(); }
    float getHumidityRaw() const { return dhtSensor.getRelativeHumidityInPercentage(); }
    int getLightPercent() const { return ldrSensor.getLightPercent(); }
    int getGasPercent() const { return gasSensor.getGasPercent(); }

    // -----------------------------------------------------------------------
    // Actualización de umbrales desde /sync/config
    // -----------------------------------------------------------------------
    void updateCachedConfig(bool linked, long newId,
                            float tempMin, float tempMax,
                            float humMin, float humMax,
                            float lightMin) {
        if (linked) {
            nodeId = (int)newId;
            cachedTempMin = tempMin;
            cachedTempMax = tempMax;
            cachedHumidityMin = humMin;
            cachedHumidityMax = humMax;
            cachedLightMin = lightMin;
            Serial.println(F("[Config] Umbrales actualizados desde backend."));
            evaluateBusinessRules();   // reacciona de inmediato a los nuevos límites
        }
    }

    // -----------------------------------------------------------------------
    // Aplicar estado deseado desde /sync/status
    // -----------------------------------------------------------------------
    void applyDesiredState(const String& buzzerMode,
                           const String& servoMode,
                           const String& ledMode) {
        BuzzerMode newBuzzer = (buzzerMode == "ON") ? BuzzerMode::ON : BuzzerMode::OFF;

        ActuatorMode newServo = ActuatorMode::AUTO;
        if (servoMode == "ON") newServo = ActuatorMode::ON;
        else if (servoMode == "OFF") newServo = ActuatorMode::OFF;

        ActuatorMode newLed = ActuatorMode::AUTO;
        if (ledMode == "ON") newLed = ActuatorMode::ON;
        else if (ledMode == "OFF") newLed = ActuatorMode::OFF;

        bool changed = false;
        if (newBuzzer != currentBuzzerMode) { currentBuzzerMode = newBuzzer; changed = true; }
        if (newServo != currentServoMode) { currentServoMode = newServo; changed = true; }
        if (newLed != currentLedMode) { currentLedMode = newLed; changed = true; }

        if (changed) {
            Serial.println(F("[Sync] Modos actualizados desde backend."));
            evaluateBusinessRules();
        }
    }

    // -----------------------------------------------------------------------
    // Envío de telemetría (llamado solo desde loop())
    // -----------------------------------------------------------------------
    void sendTelemetry() {
        if (authToken == "" || nodeId == 0) return;

        float tempC = dhtSensor.getTemperatureInCelsius();
        float humidity = dhtSensor.getRelativeHumidityInPercentage();
        int gasPct = gasSensor.getGasPercent();
        int lightPct = ldrSensor.getLightPercent();

        HTTPClient http;
        http.setTimeout(HTTP_TIMEOUT_MS);
        http.begin(secureClient, String(BASE_URL) + "/api/v1/iot/edge/sensor-readings/batch");
        http.addHeader("Content-Type", "application/json");
        http.addHeader("Authorization", "Bearer " + authToken);

        String payload = "[{\"nodeId\":" + String(nodeId)
                       + ",\"lightPercent\":" + String(lightPct)
                       + ",\"gasPercent\":" + String(gasPct)
                       + ",\"airTemperature\":" + (tempC == INVALID_READING ? "null" : String(tempC, 1))
                       + ",\"humidityPercent\":" + (humidity == INVALID_READING ? "null" : String((int)round(humidity)))
                       + "}]";

        int code = http.POST(payload);
        if (code == 201 || code == 200) {
            Serial.printf("[Telemetry] POST /edge/sensor-readings/batch -> %d\n", code);
        } else if (code == 401) {
            Serial.println(F("[Auth] Token expirado en telemetría"));
            reLogin();
        } else {
            Serial.printf("[Telemetry] Error HTTP: %d\n", code);
        }
        http.end();
    }

    // -----------------------------------------------------------------------
    // Si hay un cambio de actuador pendiente, envía el reporte (desde loop())
    // -----------------------------------------------------------------------
    void flushPendingReport() {
        if (!actuatorReportPending) return;
        actuatorReportPending = false;
        reportActuatorState();
    }

    // -----------------------------------------------------------------------
    // Actualización de pantallas LCD (llamado solo desde loop(): evita usar el
    // bus I2C desde dos tareas a la vez)
    // -----------------------------------------------------------------------
    void updateDisplays() {
        float tempC = dhtSensor.getTemperatureInCelsius();
        float humidity = dhtSensor.getRelativeHumidityInPercentage();
        int gasPct = gasSensor.getGasPercent();
        int lightPct = ldrSensor.getLightPercent();

        char row1[17], row2[17];
        snprintf(row1, sizeof(row1), "T:%.1fC H:%.1f%%",
                 (tempC == INVALID_READING ? 0.0 : tempC),
                 (humidity == INVALID_READING ? 0.0 : humidity));
        snprintf(row2, sizeof(row2), "L:%d%% A:%d%%", lightPct, gasPct);
        statusDisplay1.setLineBuffer(0, row1);
        statusDisplay1.setLineBuffer(1, row2);
        statusDisplay1.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);

        auto modeToStr = [](ActuatorMode m) -> const char* {
            return m == ActuatorMode::ON  ? "ON" :
                   m == ActuatorMode::OFF ? "OFF" : "AUTO";
        };
        const char* bzStr = currentBuzzerMode == BuzzerMode::ON ? "ON" : "OFF";

        char row3[17], row4[17];
        snprintf(row3, sizeof(row3), "Bz:%-3s Sv:%-4s", bzStr, modeToStr(currentServoMode));
        snprintf(row4, sizeof(row4), "Ld:%-4s", modeToStr(currentLedMode));
        statusDisplay2.setLineBuffer(0, row3);
        statusDisplay2.setLineBuffer(1, row4);
        statusDisplay2.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);

        Serial.printf("[Metricas] T:%.1fC H:%.1f%% L:%d%% Aire:%d%% | Bz:%s Sv:%s Ld:%s\n",
                      tempC, humidity, lightPct, gasPct,
                      bzStr, modeToStr(currentServoMode), modeToStr(currentLedMode));
    }

    // -----------------------------------------------------------------------
    // Manejo de eventos (sensores y botones) — SOLO lógica local, sin red
    // -----------------------------------------------------------------------
    void on(Event event) override {
        if (event.identifier == Button::BUTTON_PRESSED_EVENT_IDENTIFIER) {
            if (event.sourceId == BTN1_PIN) {
                currentBuzzerMode = (currentBuzzerMode == BuzzerMode::ON)
                                        ? BuzzerMode::OFF : BuzzerMode::ON;
            } else if (event.sourceId == BTN2_PIN) {
                int raw = (static_cast<int>(currentServoMode) + 1) % 3;
                currentServoMode = static_cast<ActuatorMode>(raw);
            } else if (event.sourceId == BTN3_PIN) {
                int raw = (static_cast<int>(currentLedMode) + 1) % 3;
                currentLedMode = static_cast<ActuatorMode>(raw);
            }
            evaluateBusinessRules();
        }
        else if (event.identifier == Sensor::DATA_READ_EVENT_IDENTIFIER) {
            evaluateBusinessRules();
        }
    }
};

// ---------------------------------------------------------------------------
// Re-autenticación JWT (llamada solo desde loop())
// ---------------------------------------------------------------------------
void reLogin() {
    if (WiFi.status() != WL_CONNECTED) {
        Serial.println(F("[Auth] Sin WiFi para re-autenticar."));
        return;
    }

    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.begin(secureClient, String(BASE_URL) + "/api/v1/authentication/sign-in");
    http.addHeader("Content-Type", "application/json");

    String payload = "{\"email\":\"" + String(WOKWI_EMAIL)
                   + "\",\"password\":\"" + String(WOKWI_PASSWORD) + "\"}";
    Serial.println(F("[Auth] Re-autenticando..."));
    int code = http.POST(payload);

    if (code > 0) {
        String body = http.getString();
        if (code == 200 || code == 201) {
            JsonDocument doc;
            DeserializationError err = deserializeJson(doc, body);
            if (!err) {
                authToken = doc["token"].as<String>();
                Serial.println(F("[Auth] Re-autenticación exitosa."));
            }
        } else {
            Serial.printf("[Auth] Error servidor: %d\n", code);
        }
    } else {
        Serial.printf("[Auth] Error conexión: %d\n", code);
    }
    http.end();
}

// ---------------------------------------------------------------------------
// Login inicial
// ---------------------------------------------------------------------------
bool doLogin() {
    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.begin(secureClient, String(BASE_URL) + "/api/v1/authentication/sign-in");
    http.addHeader("Content-Type", "application/json");

    String payload = "{\"email\":\"" + String(WOKWI_EMAIL)
                   + "\",\"password\":\"" + String(WOKWI_PASSWORD) + "\"}";
    Serial.println(F("[Auth] Solicitando token..."));
    int code = http.POST(payload);

    bool ok = false;
    if (code > 0) {
        String body = http.getString();
        if (code == 200 || code == 201) {
            JsonDocument doc;
            DeserializationError err = deserializeJson(doc, body);
            if (!err) {
                authToken = doc["token"].as<String>();
                Serial.println(F("[Auth] Token obtenido."));
                ok = true;
            } else {
                Serial.print(F("[Auth] Error JSON: "));
                Serial.println(err.c_str());
            }
        } else {
            Serial.printf("[Auth] Error %d: %s\n", code, body.c_str());
        }
    } else {
        Serial.printf("[Auth] Error conexión: %d\n", code);
    }
    http.end();
    return ok;
}

// ---------------------------------------------------------------------------
// GET /sync/config -> actualiza umbrales del dispositivo
// ---------------------------------------------------------------------------
void fetchConfig() {
    if (authToken == "" || pothosDevice == nullptr) return;

    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.begin(secureClient, String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/sync/config");
    http.addHeader("Authorization", "Bearer " + authToken);

    int code = http.GET();
    if (code == 200) {
        String body = http.getString();
        JsonDocument doc;
        DeserializationError err = deserializeJson(doc, body);
        if (!err) {
            bool linked = doc["linked"];
            long newId = doc["nodeId"] | 0;
            float tMin = doc["temperatureThresholdMin"] | DEFAULT_TEMP_MIN;
            float tMax = doc["temperatureThresholdMax"] | DEFAULT_TEMP_MAX;
            float hMin = doc["humidityThresholdMin"] | DEFAULT_HUMIDITY_MIN;
            float hMax = doc["humidityThresholdMax"] | DEFAULT_HUMIDITY_MAX;
            float lMin = doc["lightThresholdMin"] | DEFAULT_LIGHT_MIN;
            pothosDevice->updateCachedConfig(linked, newId, tMin, tMax, hMin, hMax, lMin);
        }
    } else if (code == 401) {
        Serial.println(F("[Auth] Token expirado en sync/config"));
        reLogin();
    } else if (code > 0) {
        Serial.printf("[Config] Error servidor: %d\n", code);
    } else {
        Serial.printf("[Config] Error conexión: %d\n", code);
    }
    http.end();
}

// ---------------------------------------------------------------------------
// GET /sync/status -> aplica el estado deseado (modos) desde el backend
// ---------------------------------------------------------------------------
void fetchStatus() {
    if (authToken == "" || pothosDevice == nullptr) return;

    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.begin(secureClient, String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/sync/status");
    http.addHeader("Authorization", "Bearer " + authToken);

    int code = http.GET();
    if (code == 200) {
        String body = http.getString();
        JsonDocument doc;
        DeserializationError err = deserializeJson(doc, body);
        if (!err) {
            String buzzer = doc["desiredState"]["buzzerMode"];
            String servo  = doc["desiredState"]["servoMode"];
            String led    = doc["desiredState"]["ledMode"];
            pothosDevice->applyDesiredState(buzzer, servo, led);
        }
    } else if (code == 401) {
        Serial.println(F("[Auth] Token expirado en sync/status"));
        reLogin();
    }
    http.end();
}

// ---------------------------------------------------------------------------
// setup()
// ---------------------------------------------------------------------------
void setup() {
    Serial.begin(SERIAL_BAUD_RATE);
    delay(1000);
    Serial.println(F("[Boot] Inicializando Pothos Care IoT..."));

    // TLS sin verificación de certificado (suficiente para el prototipo/Wokwi).
    secureClient.setInsecure();

    // ---- 1. WiFi ----
    WiFi.begin("Wokwi-GUEST", "");
    Serial.print(F("[Network] Conectando..."));
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println(F(" OK"));

    // ---- 2. Login ----
    doLogin();

    // ---- 3. Instanciar dispositivo ----
    pothosDevice = new PothosCareDevice();

    // ---- 4. Config inicial (umbrales) ----
    fetchConfig();

    delay(1000);
}

// ---------------------------------------------------------------------------
// loop() — ÚNICO punto donde se hace E/S de red
// ---------------------------------------------------------------------------
void loop() {
    static unsigned long lastSyncStatus = 0;
    static unsigned long lastSyncConfig = 0;
    static unsigned long lastTelemetry = 0;
    static unsigned long lastDisplay = 0;
    unsigned long now = millis();

    if (pothosDevice != nullptr && authToken != "") {

        // ---- /sync/status (modos deseados) ----
        if (now - lastSyncStatus >= SYNC_STATUS_INTERVAL_MS) {
            lastSyncStatus = now;
            fetchStatus();
        }

        // ---- /sync/config (umbrales) ----
        if (now - lastSyncConfig >= SYNC_CONFIG_INTERVAL_MS) {
            lastSyncConfig = now;
            fetchConfig();
        }

        // ---- Telemetría ----
        if (now - lastTelemetry >= TELEMETRY_INTERVAL_MS) {
            lastTelemetry = now;
            pothosDevice->sendTelemetry();
        }

        // ---- Reporte de actuadores pendiente ----
        pothosDevice->flushPendingReport();
    }

    // ---- Refresco de pantallas (solo aquí, para no compartir I2C entre tareas) ----
    if (pothosDevice != nullptr && (now - lastDisplay >= MAIN_LOOP_DELAY_MS)) {
        lastDisplay = now;
        pothosDevice->updateDisplays();
    }

    vTaskDelay(pdMS_TO_TICKS(MAIN_LOOP_DELAY_MS));
}
