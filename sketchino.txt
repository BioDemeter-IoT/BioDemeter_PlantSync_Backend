/**
 * @file PothosCareDevice.ino
 * @brief Sistema IoT para el cuidado de una planta Pothos usando ModestIoT asíncrono.
 *        Comunicación con backend mediante JWT Bearer token (sin Edge API Key).

#include <ModestIoT.h>
#include <Button.h>
#include <HTTPClient.h>
#include <WiFi.h>

// Variables globales de sesión
String authToken = "";

// Identificador del nodo (coincide con backend)
String nodeCode = "wokwi-esp32-002";
int nodeId = 0; // Se obtiene de GET /sync/config en setup()

// Base URL del backend
const char* BASE_URL = "https://plantsync-backend-a8c5cbd9c5bggfg3.mexicocentral-01.azurewebsites.net";

// Credenciales de la cuenta Wokwi (usuario real, login por email/password)
const char* WOKWI_EMAIL = "test@plantsync.io";
const char* WOKWI_PASSWORD = "Password123!";

// Definición de pines alineada con el diagrama Wokwi
static const int DHT_PIN = 4;
static const int GAS_A0 = 34;
static const int LDR_PIN = 32;
static const int RELAY_PIN = 5;
static const int SERVO_PIN = 18;
static const int BUZZER_PIN = 19;
static const int BTN1_PIN = 25; // Botón Buzzer
static const int BTN2_PIN = 26; // Botón Servo
static const int BTN3_PIN = 27; // Botón LED/Foco

static const uint8_t LCD1_I2C_ADDRESS = 0x27; // LCD Sensores
static const uint8_t LCD2_I2C_ADDRESS = 0x28; // LCD Actuadores

// Constantes de configuración
static const unsigned long SERIAL_BAUD_RATE = 115200;
static const unsigned long SAMPLING_INTERVAL_MS = 15;
static const unsigned long MAIN_LOOP_DELAY_MS = 500;
static const int ASYNC_ENGINE_QUEUE_DEPTH = 8;
static const int TIMER_GROUP_INDEX = 0;

// Intervalos de polling (ms)
static const unsigned long SYNC_STATUS_INTERVAL_MS = 2000;
static const unsigned long SYNC_CONFIG_INTERVAL_MS = 60000;
static const unsigned long TELEMETRY_INTERVAL_MS = 10000;

// Defaults locales cuando linked == false
static const float DEFAULT_HUMIDITY_THRESHOLD = 40.0f;
static const float DEFAULT_LIGHT_THRESHOLD = 50.0f;
static const float DEFAULT_TEMP_THRESHOLD = 15.0f;

/**
 * @brief Clase personalizada LDR (devuelve % de 0 a 100)
 */
class LdrSensor : public Sensor {
private:
    int cachedLightPercent;
    int lastRawValue;
protected:
    void processEvent(Event& event) override {
        if (event.identifier == MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER) {
            int currentRaw = analogRead(pin);
            if (abs(currentRaw - lastRawValue) > 50) {
                lastRawValue = currentRaw;
                cachedLightPercent = map(currentRaw, 0, 4095, 0, 100);
                event = Event(DATA_READ_EVENT_IDENTIFIER, pin);
            } else {
                event.identifier = -1;
            }
        }
    }
public:
    LdrSensor(int signalGpioPin, EventHandler* parentHandler = nullptr)
        : Sensor(signalGpioPin, parentHandler), cachedLightPercent(0), lastRawValue(0) {
        pinMode(pin, INPUT);
    }
    int getLightPercent() const { return cachedLightPercent; }
};

/**
 * @brief Clase personalizada GasSensor (devuelve % de contaminación de 0 a 100)
 */
class AnalogGasSensor : public Sensor {
private:
    int cachedGasPercent;
    int lastRawValue;
protected:
    void processEvent(Event& event) override {
        if (event.identifier == MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER) {
            int currentRaw = analogRead(pin);
            if (abs(currentRaw - lastRawValue) > 50) {
                lastRawValue = currentRaw;
                cachedGasPercent = map(currentRaw, 0, 4095, 0, 100);
                event = Event(DATA_READ_EVENT_IDENTIFIER, pin);
            } else {
                event.identifier = -1;
            }
        }
    }
public:
    AnalogGasSensor(int signalGpioPin, EventHandler* parentHandler = nullptr)
        : Sensor(signalGpioPin, parentHandler), cachedGasPercent(0), lastRawValue(0) {
        pinMode(pin, INPUT);
    }
    int getGasPercent() const { return cachedGasPercent; }
};

/**
 * @brief Mediador del Dispositivo de Cuidado Pothos.
 */
class PothosCareDevice : public Device {
public:
    enum class BuzzerMode { ON, OFF };
    enum class ActuatorMode { ON, OFF, AUTO };

private:
    DhtSensor dhtSensor;
    AnalogGasSensor gasSensor;
    LdrSensor ldrSensor;

    Button btnBuzzer;
    Button btnServo;
    Button btnLed;

    Led uvRelay;
    ServoMotor waterValve;
    Buzzer alarmBuzzer;

    CharacterLcdDisplay statusDisplay1; // Sensores
    CharacterLcdDisplay statusDisplay2; // Actuadores

    // Modos deseados (desde backend o botones locales)
    BuzzerMode currentBuzzerMode;
    ActuatorMode currentServoMode;
    ActuatorMode currentLedMode;

    // Estado físico real de los actuadores (después de aplicar reglas)
    bool physicalBuzzerOn;
    bool physicalServoOn;
    bool physicalLedOn;

    // Umbrales cacheados desde /sync/config
    float cachedTempMin;
    float cachedTempMax;
    float cachedHumidityMin;
    float cachedHumidityMax;
    float cachedLightMin;
    float cachedLightMax;

    // Tracking de cambios remotos para no re-evaluar innecesariamente
    BuzzerMode lastSyncedBuzzerMode;
    ActuatorMode lastSyncedServoMode;
    ActuatorMode lastSyncedLedMode;

    /**
     * @brief Envía PUT /actuators/reported con el estado físico actual.
     */
    void reportActuatorState() {
        if (authToken == "") return;

        HTTPClient http;
        String url = String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/actuators/reported";
        http.begin(url);
        http.addHeader("Content-Type", "application/json");
        http.addHeader("Authorization", "Bearer " + authToken);

        String buzzerStr = physicalBuzzerOn ? "ON" : "OFF";
        String servoStr = physicalServoOn ? "ON" : "OFF";
        String ledStr = physicalLedOn ? "ON" : "OFF";

        String payload = "{\"buzzerMode\":\"" + buzzerStr + "\",\"servoMode\":\"" + servoStr + "\",\"ledMode\":\"" + ledStr + "\"}";
        int code = http.PUT(payload);
        if (code > 0) {
            Serial.printf("[Reported] PUT /actuators/reported -> %d\n", code);
        } else {
            Serial.printf("[Reported] Error HTTP: %d\n", code);
        }
        http.end();
    }

    /**
     * @brief Evalúa las reglas de negocio y actualiza actuadores físicos.
     *        Después de aplicar cambios, reporta el estado real al backend.
     */
    void evaluateBusinessRules() {
        float tempC = dhtSensor.getTemperatureInCelsius();
        float humidity = dhtSensor.getRelativeHumidityInPercentage();
        int airQualityPercent = gasSensor.getGasPercent();
        int lightPercent = ldrSensor.getLightPercent();

        bool changed = false;

        // 1. Regla de Humedad (Válvula de agua / servo)
        bool newServoState;
        if (currentServoMode == ActuatorMode::ON) {
            waterValve.handle(ServoMotor::LOCK_TO_MAXIMUM_LIMIT_COMMAND);
            newServoState = true;
        } else if (currentServoMode == ActuatorMode::OFF) {
            waterValve.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
            newServoState = false;
        } else { // AUTO
            if (humidity > 0 && humidity < cachedHumidityMin) {
                waterValve.handle(ServoMotor::LOCK_TO_MAXIMUM_LIMIT_COMMAND);
                newServoState = true;
            } else {
                waterValve.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
                newServoState = false;
            }
        }
        if (newServoState != physicalServoOn) {
            physicalServoOn = newServoState;
            changed = true;
        }

        // 2. Regla de Luz (Foco UV / Relay)
        bool newLedState;
        if (currentLedMode == ActuatorMode::ON) {
            uvRelay.handle(Led::TURN_ON_COMMAND);
            newLedState = true;
        } else if (currentLedMode == ActuatorMode::OFF) {
            uvRelay.handle(Led::TURN_OFF_COMMAND);
            newLedState = false;
        } else { // AUTO
            if (lightPercent > cachedLightMax) {
                uvRelay.handle(Led::TURN_ON_COMMAND);
                newLedState = true;
            } else {
                uvRelay.handle(Led::TURN_OFF_COMMAND);
                newLedState = false;
            }
        }
        if (newLedState != physicalLedOn) {
            physicalLedOn = newLedState;
            changed = true;
        }

        // 3. Regla de Temperatura y Calidad de Aire (Alarma / Buzzer)
        // Regla fija: gas > 35 siempre activa la alarma cuando el modo no es OFF
        bool newBuzzerState;
        if (currentBuzzerMode == BuzzerMode::OFF) {
            alarmBuzzer.handle(Buzzer::TURN_OFF_COMMAND);
            newBuzzerState = false;
        } else { // ON
            if ((tempC != -999.0f && tempC < cachedTempMin) || airQualityPercent > 35) {
                alarmBuzzer.handle(Buzzer::TURN_ON_COMMAND);
                newBuzzerState = true;
            } else {
                alarmBuzzer.handle(Buzzer::TURN_OFF_COMMAND);
                newBuzzerState = false;
            }
        }
        if (newBuzzerState != physicalBuzzerOn) {
            physicalBuzzerOn = newBuzzerState;
            changed = true;
        }

        if (changed) {
            reportActuatorState();
        }
    }

public:
    PothosCareDevice()
        : Device(SAMPLING_INTERVAL_MS, TIMER_GROUP_INDEX),
          dhtSensor(DHT_PIN, DHT22, this),
          gasSensor(GAS_A0, this),
          ldrSensor(LDR_PIN, this),
          btnBuzzer(BTN1_PIN, 50, this),
          btnServo(BTN2_PIN, 50, this),
          btnLed(BTN3_PIN, 50, this),
          uvRelay(RELAY_PIN, false, this),
          waterValve(SERVO_PIN, 0, this),
          alarmBuzzer(BUZZER_PIN, this),
          statusDisplay1(LCD1_I2C_ADDRESS, 16, 2, true, this),
          statusDisplay2(LCD2_I2C_ADDRESS, 16, 2, true, this),
          currentBuzzerMode(BuzzerMode::ON),
          currentServoMode(ActuatorMode::AUTO),
          currentLedMode(ActuatorMode::AUTO),
          physicalBuzzerOn(false),
          physicalServoOn(false),
          physicalLedOn(false),
          cachedTempMin(DEFAULT_TEMP_THRESHOLD),
          cachedTempMax(28.0f),
          cachedHumidityMin(DEFAULT_HUMIDITY_THRESHOLD),
          cachedHumidityMax(70.0f),
          cachedLightMin(20.0f),
          cachedLightMax(DEFAULT_LIGHT_THRESHOLD),
          lastSyncedBuzzerMode(BuzzerMode::ON),
          lastSyncedServoMode(ActuatorMode::AUTO),
          lastSyncedLedMode(ActuatorMode::AUTO)
    {
        initializeAsynchronousEngine(ASYNC_ENGINE_QUEUE_DEPTH);

        appendSensorToScheduler(&dhtSensor, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&gasSensor, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&ldrSensor, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&btnBuzzer, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&btnServo, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&btnLed, Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);

        statusDisplay1.handle(CharacterLcdDisplay::TURN_BACKLIGHT_ON_COMMAND);
        statusDisplay2.handle(CharacterLcdDisplay::TURN_BACKLIGHT_ON_COMMAND);

        waterValve.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
        uvRelay.handle(Led::TURN_OFF_COMMAND);
        alarmBuzzer.handle(Buzzer::TURN_OFF_COMMAND);

        statusDisplay1.setLineBuffer(0, " Pothos Care IoT");
        statusDisplay1.setLineBuffer(1, " Iniciando...");
        statusDisplay1.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);
    }

    // --- Getters públicos para telemetría ---
    float getTemperature() const { return dhtSensor.getTemperatureInCelsius(); }
    int getLightPercent() const { return ldrSensor.getLightPercent(); }
    int getGasPercent() const { return gasSensor.getGasPercent(); }
    int getHumidity() const { return dhtSensor.getRelativeHumidityInPercentage(); }

    // --- Actualización de umbrales desde /sync/config ---
    void updateCachedConfig(bool linked, long newId, float tempMin, float tempMax,
                            float humMin, float humMax, float ltMin, float ltMax) {
        if (linked) {
            nodeId = (int)newId;
            cachedTempMin = tempMin;
            cachedTempMax = tempMax;
            cachedHumidityMin = humMin;
            cachedHumidityMax = humMax;
            cachedLightMin = ltMin;
            cachedLightMax = ltMax;
            Serial.println("[Config] Umbrales actualizados desde backend.");
        }
        // Si linked == false, mantener defaults (no se tocan las cached)
    }

    // --- Aplicar estado deseado desde /sync/status ---
    void applyDesiredState(const String& buzzerMode, const String& servoMode, const String& ledMode) {
        BuzzerMode newBuzzer = (buzzerMode == "ON") ? BuzzerMode::ON : BuzzerMode::OFF;
        ActuatorMode newServo = ActuatorMode::AUTO;
        if (servoMode == "ON") newServo = ActuatorMode::ON;
        else if (servoMode == "OFF") newServo = ActuatorMode::OFF;

        ActuatorMode newLed = ActuatorMode::AUTO;
        if (ledMode == "ON") newLed = ActuatorMode::ON;
        else if (ledMode == "OFF") newLed = ActuatorMode::OFF;

        bool modeChanged = false;
        if (newBuzzer != currentBuzzerMode) { currentBuzzerMode = newBuzzer; modeChanged = true; }
        if (newServo != currentServoMode) { currentServoMode = newServo; modeChanged = true; }
        if (newLed != currentLedMode) { currentLedMode = newLed; modeChanged = true; }

        if (modeChanged) {
            Serial.println("[Sync] Modos actualizados desde backend.");
            evaluateBusinessRules();
        }
    }

    /**
     * @brief Gestión del LCD y Puerto Serie.
     */
    void updateDisplays(bool force = false) {
        static unsigned long lastDisplayUpdate = 0;

        if (force || (millis() - lastDisplayUpdate >= 500)) {
            lastDisplayUpdate = millis();

            float tempC = dhtSensor.getTemperatureInCelsius();
            float humidity = dhtSensor.getRelativeHumidityInPercentage();
            int airQualityPercent = gasSensor.getGasPercent();
            int lightPercent = ldrSensor.getLightPercent();

            char row1[17];
            char row2[17];
            snprintf(row1, sizeof(row1), "T:%.1fC H:%.1f%%", (tempC == -999.0 ? 0.0 : tempC), (humidity == -999.0 ? 0.0 : humidity));
            snprintf(row2, sizeof(row2), "L:%d%% A:%d%%", lightPercent, airQualityPercent);
            statusDisplay1.setLineBuffer(0, row1);
            statusDisplay1.setLineBuffer(1, row2);
            statusDisplay1.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);

            auto actToStr = [](ActuatorMode m) -> const char* {
                return m == ActuatorMode::ON ? "ON" : (m == ActuatorMode::OFF ? "OFF" : "AUTO");
            };
            const char* bzStr = currentBuzzerMode == BuzzerMode::ON ? "ON" : "OFF";

            char row3[17];
            char row4[17];
            snprintf(row3, sizeof(row3), "Bz:%-3s Sv:%-4s", bzStr, actToStr(currentServoMode));
            snprintf(row4, sizeof(row4), "Ld:%-4s", actToStr(currentLedMode));
            statusDisplay2.setLineBuffer(0, row3);
            statusDisplay2.setLineBuffer(1, row4);
            statusDisplay2.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);

            Serial.printf("[Metricas] T:%.1fC H:%.1f%% L:%d%% Aire:%d%% | Modos-> Bz:%s Sv:%s Ld:%s\n",
                          tempC, humidity, lightPercent, airQualityPercent, bzStr, actToStr(currentServoMode), actToStr(currentLedMode));
        }
    }

    void on(Event event) override {
        if (event.identifier == Button::BUTTON_PRESSED_EVENT_IDENTIFIER) {
            bool changed = false;
            if (event.sourceId == BTN1_PIN) {
                currentBuzzerMode = (currentBuzzerMode == BuzzerMode::ON) ? BuzzerMode::OFF : BuzzerMode::ON;
                changed = true;
            } else if (event.sourceId == BTN2_PIN) {
                currentServoMode = static_cast<ActuatorMode>((static_cast<int>(currentServoMode) + 1) % 3);
                changed = true;
            } else if (event.sourceId == BTN3_PIN) {
                currentLedMode = static_cast<ActuatorMode>((static_cast<int>(currentLedMode) + 1) % 3);
                changed = true;
            }
            if (changed) {
                evaluateBusinessRules();
                updateDisplays(true);
                // evaluateBusinessRules ya llamó reportActuatorState si hubo cambio físico
            }
        }
        else if (event.identifier == Sensor::DATA_READ_EVENT_IDENTIFIER) {
            evaluateBusinessRules();
            updateDisplays(false);
        }
    }

    /**
     * @brief Envía telemetría al backend (POST /edge/sensor-readings/batch).
     *        Se llama periódicamente desde loop().
     */
    void sendTelemetry() {
        if (authToken == "" || nodeId == 0) return;

        float tempC = dhtSensor.getTemperatureInCelsius();
        float humidity = dhtSensor.getRelativeHumidityInPercentage();
        int airQualityPercent = gasSensor.getGasPercent();
        int lightPercent = ldrSensor.getLightPercent();

        HTTPClient http;
        String url = String(BASE_URL) + "/api/v1/iot/edge/sensor-readings/batch";
        http.begin(url);
        http.addHeader("Content-Type", "application/json");
        http.addHeader("Authorization", "Bearer " + authToken);

        String payload = "[{\"nodeId\":" + String(nodeId) +
            ",\"airTemperature\":" + (tempC == -999.0 ? "null" : String(tempC, 1)) +
            ",\"airHumidity\":" + (humidity == -999.0 ? "null" : String(humidity, 1)) +
            ",\"soilMoisture\":null" +
            ",\"gasPercent\":" + String(airQualityPercent) +
            ",\"lightIntensity\":" + String(lightPercent) +
            ",\"recordedAt\":\"\"}]";

        int code = http.POST(payload);
        if (code > 0) {
            Serial.printf("[Telemetry] POST /edge/sensor-readings/batch -> %d\n", code);
        } else {
            Serial.printf("[Telemetry] Error HTTP: %d\n", code);
        }
        http.end();
    }
};

static PothosCareDevice* pothosDevice = nullptr;


void setup() {
    Serial.begin(SERIAL_BAUD_RATE);
    delay(1000);

    Serial.println(F("[System Boot] Inicializando BioDemeter IoT..."));

    // 1. Conexión WiFi
    WiFi.begin("Wokwi-GUEST", "");
    Serial.print(F("[Network] Conectando a WiFi..."));
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println(F(" Conectado!"));

    // 2. Autenticación (login por email/password — cuenta de usuario real)
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        String url = String(BASE_URL) + "/api/v1/authentication/sign-in";
        http.begin(url);
        http.addHeader("Content-Type", "application/json");

        String jsonPayload = "{\"email\":\"" + String(WOKWI_EMAIL) + "\",\"password\":\"" + String(WOKWI_PASSWORD) + "\"}";
        Serial.println(F("[Auth] Solicitando token de acceso..."));
        int httpResponseCode = http.POST(jsonPayload);

        if (httpResponseCode > 0) {
            String responseBody = http.getString();

            if (httpResponseCode == 200 || httpResponseCode == 201) {
                JsonDocument doc;
                DeserializationError error = deserializeJson(doc, responseBody);

                if (!error) {
                    authToken = doc["token"].as<String>();
                    Serial.println(F("[Auth] Autenticación Exitosa."));
                    Serial.println(F("[Auth] Token capturado correctamente."));
                } else {
                    Serial.print(F("[Auth] Error procesando JSON: "));
                    Serial.println(error.c_str());
                }
            } else {
                Serial.printf("[Auth] Error del servidor. Código: %d\n", httpResponseCode);
                Serial.println(responseBody);
            }
        } else {
            Serial.printf("[Auth] Error en la conexión HTTP. Código: %d\n", httpResponseCode);
        }
        http.end();

        // 3. Obtener configuración inicial del nodo (/sync/config)
        if (authToken != "") {
            HTTPClient httpConfig;
            String configUrl = String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/sync/config";
            httpConfig.begin(configUrl);
            httpConfig.addHeader("Authorization", "Bearer " + authToken);

            Serial.println(F("[Config] Obteniendo configuración del nodo..."));
            int configCode = httpConfig.GET();

            if (configCode > 0) {
                String configBody = httpConfig.getString();

                if (configCode == 200) {
                    JsonDocument configDoc;
                    DeserializationError configError = deserializeJson(configDoc, configBody);

                    if (!configError) {
                        bool linked = configDoc["linked"];
                        if (linked) {
                            nodeId = configDoc["nodeId"];
                            float tempMin = configDoc["temperatureThresholdMin"];
                            float tempMax = configDoc["temperatureThresholdMax"];
                            float humMin = configDoc["humidityThresholdMin"];
                            float humMax = configDoc["humidityThresholdMax"];
                            float ltMin = configDoc["lightThresholdMin"];
                            float ltMax = configDoc["lightThresholdMax"];

                            Serial.printf("[Config] Nodo vinculado. nodeId=%d\n", nodeId);
                            Serial.printf("[Config] T:[%.1f-%.1f] H:[%.1f-%.1f] L:[%.1f-%.1f]\n",
                                          tempMin, tempMax, humMin, humMax, ltMin, ltMax);

                            // Aplicar umbrales cuando el dispositivo se cree
                            // (se aplican en el constructor vía updateCachedConfig después de crear el device)
                        } else {
                            Serial.println(F("[Config] Nodo no vinculado a planta. Usando defaults locales."));
                        }
                    } else {
                        Serial.print(F("[Config] Error procesando JSON: "));
                        Serial.println(configError.c_str());
                    }
                } else {
                    Serial.printf("[Config] Error del servidor. Código: %d\n", configCode);
                    Serial.println(configBody);
                }
            } else {
                Serial.printf("[Config] Error en la conexión HTTP. Código: %d\n", configCode);
            }
            httpConfig.end();
        }
    }

    // 4. Iniciar dispositivo principal
    pothosDevice = new PothosCareDevice();
    delay(2000);
}

void loop() {
    static unsigned long lastSyncStatus = 0;
    static unsigned long lastSyncConfig = 0;
    static unsigned long lastTelemetry = 0;

    unsigned long now = millis();

    if (pothosDevice != nullptr && authToken != "") {

        // GET /sync/status cada 2000ms (alta frecuencia)
        if (now - lastSyncStatus >= SYNC_STATUS_INTERVAL_MS) {
            lastSyncStatus = now;

            HTTPClient http;
            String url = String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/sync/status";
            http.begin(url);
            http.addHeader("Authorization", "Bearer " + authToken);

            int code = http.GET();
            if (code > 0) {
                String body = http.getString();
                JsonDocument doc;
                DeserializationError err = deserializeJson(doc, body);
                if (!err) {
                    String buzzerMode = doc["desiredState"]["buzzerMode"];
                    String servoMode = doc["desiredState"]["servoMode"];
                    String ledMode = doc["desiredState"]["ledMode"];
                    pothosDevice->applyDesiredState(buzzerMode, servoMode, ledMode);
                }
            }
            http.end();
        }

        // GET /sync/config cada 60000ms (baja frecuencia)
        if (now - lastSyncConfig >= SYNC_CONFIG_INTERVAL_MS) {
            lastSyncConfig = now;

            HTTPClient http;
            String url = String(BASE_URL) + "/api/v1/iot/nodes/" + nodeCode + "/sync/config";
            http.begin(url);
            http.addHeader("Authorization", "Bearer " + authToken);

            int code = http.GET();
            if (code > 0) {
                String body = http.getString();
                JsonDocument doc;
                DeserializationError err = deserializeJson(doc, body);
                if (!err) {
                    bool linked = doc["linked"];
                    long newId = doc["nodeId"] | 0;
                    float tempMin = doc["temperatureThresholdMin"] | DEFAULT_TEMP_THRESHOLD;
                    float tempMax = doc["temperatureThresholdMax"] | 28.0f;
                    float humMin = doc["humidityThresholdMin"] | DEFAULT_HUMIDITY_THRESHOLD;
                    float humMax = doc["humidityThresholdMax"] | 70.0f;
                    float ltMin = doc["lightThresholdMin"] | 20.0f;
                    float ltMax = doc["lightThresholdMax"] | DEFAULT_LIGHT_THRESHOLD;
                    pothosDevice->updateCachedConfig(linked, newId, tempMin, tempMax, humMin, humMax, ltMin, ltMax);
                }
            }
            http.end();
        }

        // Telemetría cada 10000ms
        if (now - lastTelemetry >= TELEMETRY_INTERVAL_MS) {
            lastTelemetry = now;
            pothosDevice->sendTelemetry();
        }
    }

    if (pothosDevice != nullptr) {
        pothosDevice->updateDisplays(false);
    }
    vTaskDelay(pdMS_TO_TICKS(MAIN_LOOP_DELAY_MS));
}
